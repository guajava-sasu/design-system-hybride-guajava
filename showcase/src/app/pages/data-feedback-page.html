Filtre et tri (JavaScript)
const table = document.querySelector("#project-table");
const filter = document.querySelector("#table-filter");
const feedback = document.querySelector("#table-feedback");
const rows = [...table.tBodies[0].rows];
let sortDirection = 1;
let sortedColumn = -1;

filter.addEventListener("input", () => {
  const query = filter.value.trim().toLocaleLowerCase();
  let count = 0;
  for (const row of rows) {
    row.hidden = !row.textContent.toLocaleLowerCase().includes(query);
    if (!row.hidden) count += 1;
  }
  feedback.textContent = query ? `${count} projet(s) affiché(s).` : "";
});

table.querySelectorAll("[data-sort]").forEach((button) => {
  button.addEventListener("click", () => {
    const column = Number(button.dataset.sort);
    if (column === sortedColumn) {
      sortDirection *= -1;
    } else {
      sortedColumn = column;
      sortDirection = 1;
    }
    for (const header of table.tHead.rows[0].cells) {
      header.removeAttribute("aria-sort");
    }
    button.closest("th").setAttribute("aria-sort", sortDirection === 1 ? "ascending" : "descending");
    rows.sort((a, b) =>
      a.cells[column].textContent.trim().localeCompare(
        b.cells[column].textContent.trim(), "fr", { numeric: true }
      ) * sortDirection
    );
    table.tBodies[0].append(...rows);
  });
});

Accordéon

Pourquoi un design system ?

Pour partager des règles fiables, améliorer l'accessibilité et éviter de recréer les mêmes composants à chaque projet.

Est-il lié à un framework ?

Non. Le socle est construit en CSS natif et peut être utilisé avec React, Vue, Angular ou du HTML simple.
